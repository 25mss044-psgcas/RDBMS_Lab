/* =========================================================
   2. CREATE TABLES
   ========================================================= */

CREATE TABLE Departments (
    department_id NUMBER PRIMARY KEY,
    department_name VARCHAR2(30)
);

CREATE TABLE Doctors (
    doctor_id NUMBER PRIMARY KEY,
    doctor_name VARCHAR2(30),
    specialization VARCHAR2(30),
    department_id NUMBER,
    salary NUMBER,
    CONSTRAINT fk_doctor_dept
        FOREIGN KEY (department_id)
        REFERENCES Departments(department_id)
);

CREATE TABLE Patients (
    patient_id NUMBER PRIMARY KEY,
    patient_name VARCHAR2(30),
    age NUMBER,
    doctor_id NUMBER,
    department_id NUMBER,
    CONSTRAINT fk_patient_doctor
        FOREIGN KEY (doctor_id)
        REFERENCES Doctors(doctor_id),
    CONSTRAINT fk_patient_dept
        FOREIGN KEY (department_id)
        REFERENCES Departments(department_id)
);

CREATE TABLE Appointments (
    appointment_id NUMBER PRIMARY KEY,
    patient_id NUMBER,
    doctor_id NUMBER,
    appointment_date DATE,
    CONSTRAINT fk_appt_patient
        FOREIGN KEY (patient_id)
        REFERENCES Patients(patient_id),
    CONSTRAINT fk_appt_doctor
        FOREIGN KEY (doctor_id)
        REFERENCES Doctors(doctor_id)
);

CREATE TABLE Employees (
    employee_id NUMBER PRIMARY KEY,
    employee_name VARCHAR2(30),
    manager_id NUMBER,
    salary NUMBER
);

CREATE OR REPLACE FUNCTION Calculate_Grade(
    p_marks IN NUMBER
)
RETURN VARCHAR2
IS
    grade VARCHAR2(2);
BEGIN
    IF p_marks >= 90 THEN
        grade := 'A';

    ELSIF p_marks >= 75 THEN
        grade := 'B';

    ELSIF p_marks >= 60 THEN
        grade := 'C';

    ELSIF p_marks >= 40 THEN
        grade := 'D';

    ELSE
        grade := 'F';
    END IF;

    RETURN grade;
END;
/

END;

CREATE OR REPLACE PROCEDURE Update_Marks(
    p_id    IN NUMBER,
    p_marks IN NUMBER
)
IS
BEGIN
    UPDATE Student
    SET Marks = p_marks
    WHERE Student_ID = p_id;

    DBMS_OUTPUT.PUT_LINE(
        'Marks updated successfully'
    );
END;

);


/* =========================================================
   30. CREATE AFTER UPDATE TRIGGER
   ========================================================= */

PROMPT ==========================================
PROMPT AFTER UPDATE TRIGGER
PROMPT ==========================================

CREATE OR REPLACE TRIGGER trg_student_audit
AFTER UPDATE ON Student
FOR EACH ROW
BEGIN

    INSERT INTO Student_Audit (
        Student_ID,
        Old_Marks,
        New_Marks,
        Updated_On
    )
    VALUES (
        :OLD.Student_ID,
        :OLD.Marks,
        :NEW.Marks,
        SYSDATE
    );

    DBMS_OUTPUT.PUT_LINE(
        'Audit record inserted successfully'
    );

END;
/
