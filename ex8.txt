CREATE OR REPLACE PROCEDURE Insert_Student (
    p_id     INT,
    p_name   VARCHAR2,
    p_gender CHAR,
    p_age    INT,
    p_email  VARCHAR2,
    p_dept   INT,
    p_marks  INT
)
IS
BEGIN
    INSERT INTO Student
    VALUES (
        p_id,
        p_name,
        p_gender,
        p_age,
        p_email,
        p_dept,
        p_marks
    );

    DBMS_OUTPUT.PUT_LINE('Student inserted successfully');
END;
/

BEGIN

        112,
        'Ravi',
        'M',
        21,
        'ravi@gmail.com',

    );
END;

CREATE OR REPLACE FUNCTION Calculate_Grade (
    marks INT
)
RETURN VARCHAR2
IS
    grade VARCHAR2(2);
BEGIN
    IF marks >= 90 THEN
        grade := 'A';
    ELSIF marks >= 75 THEN
        grade := 'B';
    ELSIF marks >= 60 THEN
        grade := 'C';
    ELSIF marks >= 40 THEN
        grade := 'D';
    ELSE
        grade := 'F';
    END IF;

    RETURN grade;
END;
/

BEGIN
    DBMS_OUTPUT.PUT_LINE(
        'Grade: ' || Calculate_Grade(82)
    );
END;
/

CREATE OR REPLACE PROCEDURE Update_Marks (
    p_id    INT,
    p_marks INT
)
IS
BEGIN
    UPDATE Student
    SET Marks = p_marks
    WHERE Student_ID = p_id;

    DBMS_OUTPUT.PUT_LINE('Marks updated successfully');
END;
/

BEGIN
    Update_Marks(101, 95);
END;
