Hospital Database – SQL Queries

Create the following tables:

Patients(patient_id, patient_name, age, doctor_id, department_id)
Doctors(doctor_id, doctor_name, specialization, department_id)
Departments(department_id, department_name)
Appointments(appointment_id, patient_id, doctor_id, appointment_date)
Employees(employee_id, employee_name, manager_id, salary)
Insert records

1. INNER JOIN
SELECT p.patient_name, d.doctor_name
FROM Patients p
INNER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

2. LEFT OUTER JOIN
SELECT p.patient_name, d.doctor_name
FROM Patients p
LEFT OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

3. RIGHT OUTER JOIN
SELECT p.patient_name, d.doctor_name
FROM Patients p
RIGHT OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

4. FULL OUTER JOIN
SELECT p.patient_name, d.doctor_name
FROM Patients p
FULL OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

5. SELF JOIN
SELECT e.employee_name AS Employee,
m.employee_name AS Manager
FROM Employees e
LEFT JOIN Employees m
ON e.manager_id = m.employee_id;

6. NESTED SUBQUERY

Find patients treated by doctors in the Cardiology department.

SELECT patient_name
FROM Patients
WHERE doctor_id IN
(
SELECT doctor_id
FROM Doctors
WHERE department_id =
(
SELECT department_id
FROM Departments
WHERE department_name = 'Cardiology'
)
);

7. CORRELATED SUBQUERY

Find doctors whose salary is greater than the average salary of their department.

SELECT doctor_name, salary, department_id
FROM Doctors d1
WHERE salary >
(
SELECT AVG(salary)
FROM Doctors d2
WHERE d1.department_id = d2.department_id
);

8. EXISTS OPERATOR

Find doctors who have at least one appointment.

SELECT doctor_name
FROM Doctors d
WHERE EXISTS
(
SELECT 1
FROM Appointments a
WHERE a.doctor_id = d.doctor_id
);

9. ANY OPERATOR

Find doctors whose salary is greater than the salary of any doctor in the Cardiology department.

SELECT doctor_name, salary
FROM Doctors
WHERE salary > ANY
(
SELECT salary
FROM Doctors
WHERE department_id =
(
SELECT department_id
FROM Departments
WHERE department_name = 'Cardiology'
)
);

10. ALL OPERATOR

Find doctors whose salary is greater than the salary of all doctors in the Cardiology department.

SELECT doctor_name, salary
FROM Doctors
WHERE salary > ALL
(
SELECT salary
FROM Doctors
WHERE department_id =
(
SELECT department_id
FROM Departments
WHERE department_name = 'Cardiology'
)
);
