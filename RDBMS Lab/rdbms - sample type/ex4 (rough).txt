alter table student add grade varchar(20);

set serveroutput on;
declare
r_count number;
begin
update student set grade;
case 
when result='pass' and avg>=90 then grade='a'
when result='pass' and avg>=80 then grade='b'
when result='pass' and avg>=70 then grade='c'
when result='pass' and avg>=60 then grade='d'
else 'better luck next time'
end;
v_count:=SQL%ROWCOUNT;
dbms_output.put_line('row updated:'||v_count);
end;
/

declare
var_row number(5);
begin
update
emp set sal=sal+100;
if SQL%NOTFOUNT THEN
dbms_output.put_line('None of the soldiers were updated');
ELSE IF SQL%FOUND THEN
var_row:=SQL%ROWCOUNT;
dbms_output.put_line('salaries for'||var_row||'employees are updated');
END IF;
END IF;
END;
/

DECLARE
cursor emp_cur IS select*from emp;
emp_rec emp%rowtype;
BEGIN
OPEN emp_cur;
LOOP
FETCH emp_cur INTO emp_rec;
EXIT WHEN emp_cur%NOTFOUND;
dbms_output.put_line(emp_rec.Emp_no||""||emp_rec.name||""||emp_rec.sal);
END LOOP;
CLOSE emp_cur;
END;
/

DECLARE

END LOOP;

END;
