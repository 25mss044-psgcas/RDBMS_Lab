CREATE TABLE Student_Audit (
    Audit_ID   INT GENERATED ALWAYS AS IDENTITY,
    Student_ID INT,
    Old_Marks  INT,
    New_Marks  INT,
    Updated_On DATE
);

CREATE OR REPLACE TRIGGER trg_student_audit
AFTER UPDATE ON Student
FOR EACH ROW
BEGIN
    INSERT INTO Student_Audit (
        Student_ID,
        Old_Marks,
        New_Marks,
        Updated_On
    )
    VALUES (
        :OLD.Student_ID,
        :OLD.Marks,
        :NEW.Marks,
        SYSDATE
    );

    DBMS_OUTPUT.PUT_LINE(
        'Audit record inserted successfully'
    );
END;
/

UPDATE Student
SET Marks = Marks + 10
WHERE Student_ID = 101;

COMMIT;

SELECT Student_ID, Student_Name, Marks
FROM Student
WHERE Student_ID = 101;

SELECT *
FROM Student_Audit;
