CREATE TABLE Departments (
    department_id   INT PRIMARY KEY,
    department_name VARCHAR2(30)
);

CREATE TABLE Doctors (
    doctor_id       INT PRIMARY KEY,
    doctor_name     VARCHAR2(30),
    specialization  VARCHAR2(30),
    department_id   INT,
    salary          INT
);

CREATE TABLE Patients (
    patient_id      INT PRIMARY KEY,
    patient_name    VARCHAR2(30),
    age             INT,
    doctor_id       INT,
    department_id   INT
);

CREATE TABLE Appointments (
    appointment_id   INT PRIMARY KEY,
    patient_id       INT,
    doctor_id        INT,
    appointment_date DATE
);

CREATE TABLE Employees (
    employee_id   INT PRIMARY KEY,
    employee_name VARCHAR2(30),
    manager_id    INT,
    salary        INT
);

SELECT *
FROM Employees;

SELECT p.patient_name, d.doctor_name
FROM Patients p
INNER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

SELECT p.patient_name, d.doctor_name
FROM Patients p
LEFT OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

SELECT p.patient_name, d.doctor_name
FROM Patients p
RIGHT OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

SELECT p.patient_name, d.doctor_name
FROM Patients p
FULL OUTER JOIN Doctors d
ON p.doctor_id = d.doctor_id;

SELECT e.employee_name AS Employee,
       m.employee_name AS Manager
FROM Employees e
LEFT JOIN Employees m
ON e.manager_id = m.employee_id;

SELECT patient_name
FROM Patients
WHERE doctor_id IN
(
    SELECT doctor_id
    FROM Doctors
    WHERE department_id =
    (
        SELECT department_id
        FROM Departments
        WHERE department_name = 'Cardiology'
    )
);

SELECT doctor_name, salary, department_id
FROM Doctors d1
WHERE salary >
(
    SELECT AVG(salary)
    FROM Doctors d2
    WHERE d1.department_id = d2.department_id
);

SELECT doctor_name
FROM Doctors d
WHERE EXISTS
(
    SELECT 1
    FROM Appointments a
    WHERE a.doctor_id = d.doctor_id
);

SELECT doctor_name, salary
FROM Doctors
WHERE salary > ANY
(
    SELECT salary
    FROM Doctors
    WHERE department_id =
    (
        SELECT department_id
        FROM Departments
        WHERE department_name = 'Cardiology'
    )
);

SELECT doctor_name, salary
FROM Doctors
WHERE salary > ALL
(
    SELECT salary
    FROM Doctors
    WHERE department_id =
    (
        SELECT department_id
        FROM Departments
        WHERE department_name = 'Cardiology'
    )
);
